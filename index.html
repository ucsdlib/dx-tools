Quick start

Include Google Fonts and all five CSS files. The FOUC prevention snippet must go in <head> before the stylesheets.

html 
<!-- FOUC prevention: must be first in <head>, before any stylesheet -->
<script>const t=localStorage.getItem('dx-theme');document.documentElement.setAttribute('data-theme',t||'light');</script>

<!-- Fonts (self-hosted — no external requests) -->
<link rel="stylesheet" href="css/fonts.css">

<!-- Design system -->
<link rel="stylesheet" href="css/tokens.css">
<link rel="stylesheet" href="css/frame.css">
<link rel="stylesheet" href="css/components.css">
<link rel="stylesheet" href="css/preview.css">
<link rel="stylesheet" href="css/code.css">

<!-- Icons + theme toggle (self-hosted Lucide) -->
<script src="js/lucide.min.js"></script>
<script src="js/theme.js" defer></script>
<script src="js/code-copy.js" defer></script>
